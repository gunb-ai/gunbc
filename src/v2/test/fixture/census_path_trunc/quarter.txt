fn is_keyword_text(text: String) -> Bool {
  match lookup(dag_keyword_set, key: text) {
    Holds { value: _ } => true
    Violates { diagnostic: _ } => false
  }
}

type TokenizerState {
  pos: Int
  tokens: List<Token>
  interp_depth: List<Int>
}

type TokPos {
  pos: Int
  interp_depth: List<Int>
}

type ScanResult {
  pos: Int
  token: Token
  interp_depth: List<Int>
}

type SourceRef {
  file: String
  text: String
  source_chars: List<Int>
}

fn make_token(text: String, span: SourceSpan, shape: TokenShape) -> Token {
  Token { text: text, span: span, shape: shape }
}

fn source_char(source: SourceRef, pos: Int) -> String {
  from_code_point(cp: source.source_chars[pos])
}

fn source_code_point(source: SourceRef, pos: Int) -> Int {
  let _ = record_source_chars_index_lookup()
  source.source_chars[pos]
}

fn source_len(source: SourceRef) -> Int {
  count(source.source_chars)
}

fn source_substring(source: SourceRef, start: Int, end: Int) -> String {
  chars_to_string(chars: source.source_chars, start: start, end: end)
}

fn source_scan_while(source: SourceRef, start: Int, pred: fn(Int) -> Bool) -> Int {
  if start >= source_len(source: source) { source_len(source: source) }
  else if pred(ch: source.source_chars[start]) {
    source_scan_while(source: source, start: start + 1, pred: pred)
  } else { start }
}

fn source_skip_ws(source: SourceRef, start: Int) -> Int {
  if start >= source_len(source: source) { start }
  else {
    let ch = source.source_chars[start]
    if ch == 32 || ch == 9 { source_skip_ws(source: source, start: start + 1) }
    else { start }
  }
}

fn source_scan_to_eol(source: SourceRef, start: Int) -> Int {
  if start >= source_len(source: source) { source_len(source: source) }
  else if source.source_chars[start] == 10 { start }
  else { source_scan_to_eol(source: source, start: start + 1) }
}

fn tokenize(source: String, file: String) -> List<Token> {
  let c = chars(source)
  let src = SourceRef { file: file, text: source, source_chars: c }
  let initial = TokPos { pos: 0, interp_depth: [] }
  let final_state = tokenize_loop(source: src, tokens: [], pos: initial, fuel: source_len(source: src) + 1)
  let eof_span = make_file_span(file: src.file, start: final_state.pos, end: final_state.pos)
  list_push(final_state.tokens, make_token(text: "", span: eof_span, shape: ShEof))
}

fn scan_next_token(source: SourceRef, pos: TokPos) -> ScanResult {
  let ch = source_code_point(source: source, pos: pos.pos)

  if ch == 10 {
    return ScanResult {
      pos: pos.pos + 1,
      token: make_token(text: "\n", span: make_file_span(file: source.file, start: pos.pos, end: pos.pos + 1), shape: ShNewline),
      interp_depth: pos.interp_depth
    }
  }

  if ch == 125 && count(pos.interp_depth) > 0 {
    let top = last(pos.interp_depth).value
    if top == 0 {
      let popped = drop_last(stack: pos.interp_depth)
      let cont_pos = TokPos { pos: pos.pos + 1, interp_depth: popped }
      return scan_str_cont(source: source, pos: cont_pos, span_start: pos.pos)
    } else {
      return ScanResult {
        pos: pos.pos + 1,
        token: make_token(text: "}", span: make_file_span(file: source.file, start: pos.pos, end: pos.pos + 1), shape: ShRBrace),
        interp_depth: replace_last(stack: pos.interp_depth, value: top - 1)
      }
    }
  }

  scan_token(source: source, pos: pos, ch: ch)
}
