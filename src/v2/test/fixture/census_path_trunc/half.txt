fn is_keyword_text(text: String) -> Bool {
  match lookup(dag_keyword_set, key: text) {
    Holds { value: _ } => true
    Violates { diagnostic: _ } => false
  }
}

data single_punct: Map<String, TokenShape> = {
  "(": ShLParen, ")": ShRParen,
  "[": ShLBracket, "]": ShRBracket,
  ":": ShColon, ",": ShComma, ".": ShDot,
  "+": ShPlus, "*": ShStar, "%": ShPercent, "/": ShSlash,
  "^": ShCaret
}

type TokenizerState {
  pos: Int
  tokens: List<Token>
  interp_depth: List<Int>
}

type TokPos {
  pos: Int
  interp_depth: List<Int>
}

type ScanResult {
  pos: Int
  token: Token
  interp_depth: List<Int>
}

type SourceRef {
  file: String
  text: String
  source_chars: List<Int>
}

fn make_token(text: String, span: SourceSpan, shape: TokenShape) -> Token {
  Token { text: text, span: span, shape: shape }
}

fn source_char(source: SourceRef, pos: Int) -> String {
  from_code_point(cp: source.source_chars[pos])
}

fn source_code_point(source: SourceRef, pos: Int) -> Int {
  let _ = record_source_chars_index_lookup()
  source.source_chars[pos]
}

fn source_len(source: SourceRef) -> Int {
  count(source.source_chars)
}

fn source_substring(source: SourceRef, start: Int, end: Int) -> String {
  chars_to_string(chars: source.source_chars, start: start, end: end)
}

fn source_scan_while(source: SourceRef, start: Int, pred: fn(Int) -> Bool) -> Int {
  if start >= source_len(source: source) { source_len(source: source) }
  else if pred(ch: source.source_chars[start]) {
    source_scan_while(source: source, start: start + 1, pred: pred)
  } else { start }
}

fn source_skip_ws(source: SourceRef, start: Int) -> Int {
  if start >= source_len(source: source) { start }
  else {
    let ch = source.source_chars[start]
    if ch == 32 || ch == 9 { source_skip_ws(source: source, start: start + 1) }
    else { start }
  }
}

fn source_scan_to_eol(source: SourceRef, start: Int) -> Int {
  if start >= source_len(source: source) { source_len(source: source) }
  else if source.source_chars[start] == 10 { start }
  else { source_scan_to_eol(source: source, start: start + 1) }
}

fn tokenize(source: String, file: String) -> List<Token> {
  let c = chars(source)
  let src = SourceRef { file: file, text: source, source_chars: c }
  let initial = TokPos { pos: 0, interp_depth: [] }
  let final_state = tokenize_loop(source: src, tokens: [], pos: initial, fuel: source_len(source: src) + 1)
  let eof_span = make_file_span(file: src.file, start: final_state.pos, end: final_state.pos)
  list_push(final_state.tokens, make_token(text: "", span: eof_span, shape: ShEof))
}

fn scan_next_token(source: SourceRef, pos: TokPos) -> ScanResult {
  let ch = source_code_point(source: source, pos: pos.pos)

  if ch == 10 {
    return ScanResult {
      pos: pos.pos + 1,
      token: make_token(text: "\n", span: make_file_span(file: source.file, start: pos.pos, end: pos.pos + 1), shape: ShNewline),
      interp_depth: pos.interp_depth
    }
  }

  if ch == 125 && count(pos.interp_depth) > 0 {
    let top = last(pos.interp_depth).value
    if top == 0 {
      let popped = drop_last(stack: pos.interp_depth)
      let cont_pos = TokPos { pos: pos.pos + 1, interp_depth: popped }
      return scan_str_cont(source: source, pos: cont_pos, span_start: pos.pos)
    } else {
      return ScanResult {
        pos: pos.pos + 1,
        token: make_token(text: "}", span: make_file_span(file: source.file, start: pos.pos, end: pos.pos + 1), shape: ShRBrace),
        interp_depth: replace_last(stack: pos.interp_depth, value: top - 1)
      }
    }
  }

  scan_token(source: source, pos: pos, ch: ch)
}

fn tokenize_loop(source: SourceRef, tokens: List<Token>, pos: TokPos, fuel: Int) -> TokenizerState {
  let s = skip_spaces(source: source, pos: pos)
  if s.pos >= source_len(source: source) {
    return TokenizerState { pos: s.pos, tokens: tokens, interp_depth: s.interp_depth }
  }
  let result = scan_next_token(source: source, pos: s)
  tokenize_loop(source: source, tokens: list_push(tokens, result.token), pos: TokPos { pos: result.pos, interp_depth: result.interp_depth }, fuel: fuel - 1)
}

fn scan_token(source: SourceRef, pos: TokPos, ch: Int) -> ScanResult {
  if ch == 34 { return scan_string(source: source, pos: pos) }
  if is_digit(ch: ch) { return scan_number(source: source, pos: pos) }
  if is_ident_start(ch: ch) { return scan_ident(source: source, pos: pos) }

  let next_ch = if pos.pos + 1 < source_len(source: source) {
    source_code_point(source: source, pos: pos.pos + 1)
  } else { 0 }

  if ch == 61 && next_ch == 62 { return emit(pos: pos, shape: ShFatArrow, text: "=>", len: 2, file: source.file) }
  if ch == 45 && next_ch == 62 { return emit(pos: pos, shape: ShArrow, text: "->", len: 2, file: source.file) }
  if ch == 61 && next_ch == 61 { return emit(pos: pos, shape: ShEqEq, text: "==", len: 2, file: source.file) }
  if ch == 33 && next_ch == 61 { return emit(pos: pos, shape: ShNe, text: "!=", len: 2, file: source.file) }
  if ch == 60 && next_ch == 61 { return emit(pos: pos, shape: ShLe, text: "<=", len: 2, file: source.file) }
  if ch == 62 && next_ch == 61 { return emit(pos: pos, shape: ShGe, text: ">=", len: 2, file: source.file) }
  if ch == 38 && next_ch == 38 { return emit(pos: pos, shape: ShAnd, text: "&&", len: 2, file: source.file) }
  if ch == 124 && next_ch == 124 { return emit(pos: pos, shape: ShOr, text: "||", len: 2, file: source.file) }
  if ch == 124 && next_ch == 62 { return emit(pos: pos, shape: ShPipeArrow, text: "|>", len: 2, file: source.file) }
  if ch == 124 { return emit(pos: pos, shape: ShPipe, text: "|", len: 1, file: source.file) }
  if ch == 63 && next_ch == 63 { return emit(pos: pos, shape: ShNullCoalesce, text: "??", len: 2, file: source.file) }
  if ch == 46 && next_ch == 46 { return emit(pos: pos, shape: ShDotDot, text: "..", len: 2, file: source.file) }

  if ch == 61 { return emit(pos: pos, shape: ShEq, text: "=", len: 1, file: source.file) }
  if ch == 60 { return emit(pos: pos, shape: ShLt, text: "<", len: 1, file: source.file) }
  if ch == 62 { return emit(pos: pos, shape: ShGt, text: ">", len: 1, file: source.file) }
  if ch == 45 { return emit(pos: pos, shape: ShMinus, text: "-", len: 1, file: source.file) }
  if ch == 33 { return emit(pos: pos, shape: ShBang, text: "!", len: 1, file: source.file) }
  if ch == 63 { return emit(pos: pos, shape: ShQuestion, text: "?", len: 1, file: source.file) }

  if ch == 123 {
    let new_depth = if count(pos.interp_depth) > 0 {
      replace_last(stack: pos.interp_depth, value: last(pos.interp_depth).value + 1)
    } else {
      pos.interp_depth
    }
    let tok = make_token(text: "{", span: make_file_span(file: source.file, start: pos.pos, end: pos.pos + 1), shape: ShLBrace)
    return ScanResult {
      pos: pos.pos + 1, token: tok, interp_depth: new_depth
    }
  }

  if ch == 125 {
    let tok = make_token(text: "}", span: make_file_span(file: source.file, start: pos.pos, end: pos.pos + 1), shape: ShRBrace)
    return ScanResult {
      pos: pos.pos + 1, token: tok, interp_depth: pos.interp_depth
    }
  }

  let ch_text = source_char(source: source, pos: pos.pos)
  match lookup(single_punct, key: ch_text) {
    Holds { value: sh } => emit(pos: pos, shape: sh, text: ch_text, len: 1, file: source.file)
    Violates { diagnostic: _ } => emit(pos: pos, shape: ShUnknown, text: ch_text, len: 1, file: source.file)
  }
}

fn emit(pos: TokPos, shape: TokenShape, text: String, len: Int, file: String) -> ScanResult {
  let token = make_token(text: text, span: make_file_span(file: file, start: pos.pos, end: pos.pos + len), shape: shape)
  ScanResult {
    pos: pos.pos + len,
    token: token,
    interp_depth: pos.interp_depth
  }
}
