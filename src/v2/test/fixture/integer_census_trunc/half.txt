fn integer_standard_integer_type_decode_invalid_diagnostic(source_carrier: Node) -> Diagnostic {

    at: node_locus(node: source_carrier),
    correction: Unavailable { reason: ExternalContractUnknown }
  }
}

fn decimal_digit_value(d: DecimalDigit) -> Int {
  match d {
    D0 => 0
    D1 => 1
    D2 => 2
    D3 => 3
    D4 => 4
    D5 => 5
    D6 => 6
    D7 => 7
    D8 => 8
    D9 => 9
  }
}

fn decimal_digit_compare(left: DecimalDigit, right: DecimalDigit) -> Ordering {
  int_compare(a: decimal_digit_value(d: left), b: decimal_digit_value(d: right))
}

fn decimal_digit_list_compare(left: FreeMonoid<DecimalDigit>, right: FreeMonoid<DecimalDigit>) -> Ordering {
  match left {
    Empty => match right {
      Empty => Equal
      Cons { head: _, tail: _ } => Less
    }
    Cons { head: left_head, tail: left_tail } => match right {
      Empty => Greater
      Cons { head: right_head, tail: right_tail } =>
        match decimal_digit_compare(left: left_head, right: right_head) {
          Equal => decimal_digit_list_compare(left: left_tail, right: right_tail)
          Less => Less
          Greater => Greater
        }
    }
  }
}

fn decimal_nonzero_magnitude_compare(left: DecimalNonZeroMagnitude, right: DecimalNonZeroMagnitude) -> Ordering {
  let left_len = 1 + length(xs: left.tail)
  let right_len = 1 + length(xs: right.tail)
  if left_len < right_len {
    Less
  } else if left_len > right_len {
    Greater
  } else {
    match decimal_digit_compare(
      left: integer_nonzero_decimal_digit_widen(digit: left.lead),
      right: integer_nonzero_decimal_digit_widen(digit: right.lead)
    ) {
      Equal => decimal_digit_list_compare(left: left.tail, right: right.tail)
      Less => Less
      Greater => Greater
    }
  }
}

fn ordering_invert(ordering: Ordering) -> Ordering {
  match ordering {
    Less => Greater
    Equal => Equal
    Greater => Less
  }
}

type StandardIntegerType
  = Int128
  | Int64
  | Int32
  | Int16
  | Int8
  | UInt128
  | UInt64
  | UInt32
  | UInt16
  | UInt8

fn integer_standard_integer_type_interval_spec(standard_integer_type: StandardIntegerType) -> IntegerIntervalSpec {
  match standard_integer_type {
    Int128 => int128_interval_spec()
    Int64 => int64_interval_spec()
    Int32 => int32_interval_spec()
    Int16 => int16_interval_spec()
    Int8 => int8_interval_spec()
    UInt128 => uint128_interval_spec()
    UInt64 => uint64_interval_spec()
    UInt32 => uint32_interval_spec()
    UInt16 => uint16_interval_spec()
    UInt8 => uint8_interval_spec()
  }
}

fn integer_standard_integer_type_from_node(node: Node) -> Outcome<StandardIntegerType> {
  if node == integer_interval_spec_node(spec: int128_interval_spec()) {
    Accepted { value: Int128, diagnostics: None }
  } else if node == integer_interval_spec_node(spec: int64_interval_spec()) {
    Accepted { value: Int64, diagnostics: None }
  } else if node == integer_interval_spec_node(spec: int32_interval_spec()) {
    Accepted { value: Int32, diagnostics: None }
  } else if node == integer_interval_spec_node(spec: int16_interval_spec()) {
    Accepted { value: Int16, diagnostics: None }
  } else if node == integer_interval_spec_node(spec: int8_interval_spec()) {
    Accepted { value: Int8, diagnostics: None }
  } else if node == integer_interval_spec_node(spec: uint128_interval_spec()) {
    Accepted { value: UInt128, diagnostics: None }
  } else if node == integer_interval_spec_node(spec: uint64_interval_spec()) {
    Accepted { value: UInt64, diagnostics: None }
  } else if node == integer_interval_spec_node(spec: uint32_interval_spec()) {
    Accepted { value: UInt32, diagnostics: None }
  } else if node == integer_interval_spec_node(spec: uint16_interval_spec()) {
    Accepted { value: UInt16, diagnostics: None }
  } else if node == integer_interval_spec_node(spec: uint8_interval_spec()) {
    Accepted { value: UInt8, diagnostics: None }
  } else {
    Rejected {
      diagnostics: diagnostics_singleton(
        d: integer_standard_integer_type_decode_invalid_diagnostic(source_carrier: node)
      )
    }
  }
}

fn integer_pointer_width_bound_compare(
  left_signedness: Signedness,
  left_endpoint: IntegerRangeEndpoint,
  right_signedness: Signedness,
  right_endpoint: IntegerRangeEndpoint
) -> Ordering {
  match left_signedness {
    Signed => match right_signedness {
      Signed => match left_endpoint {
        IntegerRangeInclusiveMin => match right_endpoint {
          IntegerRangeInclusiveMin => Equal
          IntegerRangeInclusiveMax => Less
        }
        IntegerRangeInclusiveMax => match right_endpoint {
          IntegerRangeInclusiveMin => Greater
          IntegerRangeInclusiveMax => Equal
        }
      }
      Unsigned => Less
    }
    Unsigned => match right_signedness {
      Signed => Greater
      Unsigned => match left_endpoint {
        IntegerRangeInclusiveMin => match right_endpoint {
          IntegerRangeInclusiveMin => Equal
          IntegerRangeInclusiveMax => Less
        }
        IntegerRangeInclusiveMax => match right_endpoint {
          IntegerRangeInclusiveMin => Greater
          IntegerRangeInclusiveMax => Equal
        }
      }
    }
  }
}

fn integer_concrete_zero_compare_to_pointer_width(
  right_signedness: Signedness,
  right_endpoint: IntegerRangeEndpoint
) -> Ordering {
  match right_signedness {
    Signed => match right_endpoint {
      IntegerRangeInclusiveMin => Greater
      IntegerRangeInclusiveMax => Less
    }
    Unsigned => match right_endpoint {
      IntegerRangeInclusiveMin => Equal
      IntegerRangeInclusiveMax => Less
    }
  }
}

fn integer_concrete_bound_compare_to_pointer_width(
  left_sign: IntegerBoundSign,
  right_signedness: Signedness,
  right_endpoint: IntegerRangeEndpoint
) -> Ordering {
  match right_signedness {
    Signed => match right_endpoint {
      IntegerRangeInclusiveMin => Greater
      IntegerRangeInclusiveMax => Less
    }
    Unsigned => match right_endpoint {
      IntegerRangeInclusiveMin => match left_sign {
        IntegerBoundNegative => Less
        IntegerBoundNonNegative => Greater
      }
      IntegerRangeInclusiveMax => Less
    }
  }
}

fn integer_interval_bound_compare(left: IntegerIntervalBound, right: IntegerIntervalBound) -> Ordering {
  match left {
    IntegerConcreteZero => match right {
      IntegerConcreteZero => Equal
      IntegerConcreteBound { sign, magnitude: _ } => match sign {
        IntegerBoundNegative => Greater
        IntegerBoundNonNegative => Less
      }
      IntegerPointerWidthBound { signedness, endpoint } =>
        integer_concrete_zero_compare_to_pointer_width(
          right_signedness: signedness,
          right_endpoint: endpoint
        )
    }
    IntegerConcreteBound { sign: left_sign, magnitude: left_magnitude } => match right {
      IntegerConcreteZero => match left_sign {
        IntegerBoundNegative => Less
        IntegerBoundNonNegative => Greater
      }
      IntegerConcreteBound { sign: right_sign, magnitude: right_magnitude } => match left_sign {
        IntegerBoundNegative => match right_sign {
          IntegerBoundNegative => match decimal_nonzero_magnitude_compare(left: left_magnitude, right: right_magnitude) {
            Equal => Equal
            Less => Greater
            Greater => Less
          }
          IntegerBoundNonNegative => Less
        }
        IntegerBoundNonNegative => match right_sign {
          IntegerBoundNegative => Greater
          IntegerBoundNonNegative => decimal_nonzero_magnitude_compare(left: left_magnitude, right: right_magnitude)
        }
      }
      IntegerPointerWidthBound { signedness, endpoint } =>
        integer_concrete_bound_compare_to_pointer_width(
          left_sign: left_sign,
          right_signedness: signedness,
          right_endpoint: endpoint
        )
    }
    IntegerPointerWidthBound { signedness: left_signedness, endpoint: left_endpoint } =>
      match right {
        IntegerConcreteZero =>
          ordering_invert(ordering: integer_concrete_zero_compare_to_pointer_width(
            right_signedness: left_signedness,
            right_endpoint: left_endpoint
          ))
        IntegerConcreteBound { sign: right_sign, magnitude: right_magnitude } =>
          ordering_invert(ordering: integer_concrete_bound_compare_to_pointer_width(
            left_sign: right_sign,
            right_signedness: left_signedness,
            right_endpoint: left_endpoint
          ))
        IntegerPointerWidthBound { signedness: right_signedness, endpoint: right_endpoint } =>
          integer_pointer_width_bound_compare(
            left_signedness: left_signedness,
            left_endpoint: left_endpoint,
            right_signedness: right_signedness,
            right_endpoint: right_endpoint
          )
      }
  }
}

fn integer_interval_bound_less_or_equal(left: IntegerIntervalBound, right: IntegerIntervalBound) -> Bool {
  match integer_interval_bound_compare(left: left, right: right) {
    Less => true
    Equal => true
    Greater => false
  }
}

fn integer_interval_bound_is_pointer_width(bound: IntegerIntervalBound) -> Bool {
  match bound {
    IntegerPointerWidthBound { signedness: _, endpoint: _ } => true
    IntegerConcreteZero => false
    IntegerConcreteBound { sign: _, magnitude: _ } => false
  }
}

fn integer_interval_spec_is_pointer_width(spec: IntegerIntervalSpec) -> Bool {
  integer_interval_bound_is_pointer_width(bound: spec.inclusive_min)
    && integer_interval_bound_is_pointer_width(bound: spec.inclusive_max)
}

fn integer_interval_spec_is_concrete(spec: IntegerIntervalSpec) -> Bool {
  !integer_interval_bound_is_pointer_width(bound: spec.inclusive_min)
    && !integer_interval_bound_is_pointer_width(bound: spec.inclusive_max)
}
