fn decimal_digit_value(d: DecimalDigit) -> Int {
  match d {
    D0 => 0
    D1 => 1
    D2 => 2
    D3 => 3
    D4 => 4
    D5 => 5
    D6 => 6
    D7 => 7
    D8 => 8
    D9 => 9
  }
}

fn decimal_digit_compare(left: DecimalDigit, right: DecimalDigit) -> Ordering {
  int_compare(a: decimal_digit_value(d: left), b: decimal_digit_value(d: right))
}

fn decimal_digit_list_compare(left: FreeMonoid<DecimalDigit>, right: FreeMonoid<DecimalDigit>) -> Ordering {
  match left {
    Empty => match right {
      Empty => Equal
      Cons { head: _, tail: _ } => Less
    }
    Cons { head: left_head, tail: left_tail } => match right {
      Empty => Greater
      Cons { head: right_head, tail: right_tail } =>
        match decimal_digit_compare(left: left_head, right: right_head) {
          Equal => decimal_digit_list_compare(left: left_tail, right: right_tail)
          Less => Less
          Greater => Greater
        }
    }
  }
}

fn decimal_nonzero_magnitude_compare(left: DecimalNonZeroMagnitude, right: DecimalNonZeroMagnitude) -> Ordering {
  let left_len = 1 + length(xs: left.tail)
  let right_len = 1 + length(xs: right.tail)
  if left_len < right_len {
    Less
  } else if left_len > right_len {
    Greater
  } else {
    match decimal_digit_compare(
      left: integer_nonzero_decimal_digit_widen(digit: left.lead),
      right: integer_nonzero_decimal_digit_widen(digit: right.lead)
    ) {
      Equal => decimal_digit_list_compare(left: left.tail, right: right.tail)
      Less => Less
      Greater => Greater
    }
  }
}

fn ordering_invert(ordering: Ordering) -> Ordering {
  match ordering {
    Less => Greater
    Equal => Equal
    Greater => Less
  }
}
