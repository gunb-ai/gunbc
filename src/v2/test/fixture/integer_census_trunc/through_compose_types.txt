  Group,
  length,

type Int8 = Compose<Int, MachineWidth<Word8>>
type Int16 = Compose<Int, MachineWidth<Word16>>
type Int32 = Compose<Int, MachineWidth<Word32>>

type Int128 = Compose<Int, MachineWidth<Word128>>

type UInt8 = Compose<UInt, MachineWidth<Word8>>

type UInt32 = Compose<UInt, MachineWidth<Word32>>
type UInt64 = Compose<UInt, MachineWidth<Word64>>
type UInt128 = Compose<UInt, MachineWidth<Word128>>

type IntPlatform = Compose<Int, MachineWidth<PointerWidth>>
type UIntPlatform = Compose<UInt, MachineWidth<PointerWidth>>
