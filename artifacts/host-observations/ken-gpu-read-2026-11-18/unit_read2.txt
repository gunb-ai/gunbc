### t10-power-cap.service
[Unit]
Description=Cap every Tesla T10 (three on ken) at its minimum power limit
After=systemd-modules-load.service
[Service]
Type=oneshot
RemainAfterExit=yes
ExecStartPre=/sbin/modprobe nvidia
ExecStart=/usr/local/sbin/t10-power-cap.sh
[Install]
WantedBy=multi-user.target
### ken-strata.service
[Unit]
Description=Strata engine on ken: one server over all three Tesla T10s (strata-main 8c5fafc1 sm_75, IQ3_S, layer split auto, port 18081)
After=network-online.target t10-power-cap.service
Wants=network-online.target
Requires=t10-power-cap.service

[Service]
User=ubuntu
WorkingDirectory=/home/ubuntu/strata-test/strata-main
Environment=PYTHONUNBUFFERED=1
ExecStart=/home/ubuntu/strata-test/strata/.venv/bin/python serve/server.py --engine strata --config /home/ubuntu/strata-pool/cfg-ken.json --host 0.0.0.0 --port 18081
# The expert GGUF is mmapped and its page cache is charged here, so the cap leaves room for it (125 GB host).
MemoryMax=112G
MemorySwapMax=2G
Restart=on-failure
RestartSec=20
TimeoutStartSec=420

[Install]
WantedBy=multi-user.target
