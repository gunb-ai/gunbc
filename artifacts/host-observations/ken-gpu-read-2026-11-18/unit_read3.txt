### t10-power-cap.sh
#!/bin/sh
# Cap every NVIDIA GPU at its own minimum power limit (operator 2026-11-15: "lower their power as low as possible"), read back, fail if not applied.
# ken carries three Tesla T10s since 2026-11-17 (SLOT4 01:00.0, SLOT6 81:00.0, SLOT2 c1:00.0); fewer than EXPECT fails the unit.
set -eu
EXPECT=3
nvidia-smi -pm 1 >/dev/null
n=0
for i in $(nvidia-smi --query-gpu=index --format=csv,noheader); do
  min=$(nvidia-smi -i $i --query-gpu=power.min_limit --format=csv,noheader,nounits | cut -d. -f1)
  nvidia-smi -i $i -pl $min >/dev/null
  got=$(nvidia-smi -i $i --query-gpu=power.limit --format=csv,noheader,nounits | cut -d. -f1)
  echo "gpu $i ($(nvidia-smi -i $i --query-gpu=pci.bus_id --format=csv,noheader)): limit $got W (min $min W)"
  [ "$got" = "$min" ] || { echo "gpu $i readback $got != $min" >&2; exit 1; }
  n=$((n+1))
done
[ $n -eq $EXPECT ] || { echo "expected $EXPECT GPUs, saw $n" >&2; exit 1; }
