module test.claim.zz_duct_golden_scratch

import std.types { String, List }
import product.printed_chassis.printable_assembly { printed_assembly_roots, printed_assembly_root_wire, printable_manifest, PrintableAssemblyReady, PrintableAssemblyRefused }
import product.printed_chassis.printable_assembly_selection { select_printable_assembly }
import extdeps.languages.json.emit { serialize_json }
import gunbc.cli_wire { CliWireResponse, CliWirePrintable }
import std.process { ExitSuccess }

fn main() -> CliWireResponse {
  let lines = map(filter(printed_assembly_roots, r => starts_with(s: printed_assembly_root_wire(root: r), prefix: "a16")), r => join(["### ", printed_assembly_root_wire(root: r), "\n", match select_printable_assembly(root: r) {
    PrintableAssemblyReady { assembly: a } => serialize_json(v: printable_manifest(a: a))
    PrintableAssemblyRefused { reason: why } => join(["REFUSED ", why], "")
  }], ""))
  CliWirePrintable { bytes: join(lines, "\n"), exit: ExitSuccess }
}
