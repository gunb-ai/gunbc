==== acwatch.sh (sha256 c9c2a5ac2a3327cc1be9072a1a3f991d982a2d5a831bd64ff3e585cac4e1f3ca)
#!/bin/bash
cd ~/mtc1-32dimm
B="ipmitool -I lanplus -H 192.168.1.228 -U admin -P admin"
M=10-ac-meta.txt
ts(){ date -u +%Y-%m-%dT%H:%M:%S.%3NZ; }
echo "$(ts) watcher armed; waiting for BMC to GO DOWN first" > $M
# phase 1: wait for the controller to actually disappear (max 30 min)
for i in $(seq 1 1800); do
  if ! $B mc info >/dev/null 2>&1; then echo "$(ts) BMC down (ipmi no answer) after ${i}s" >> $M; break; fi
  sleep 1
done
ip neigh show 192.168.1.228 >> $M 2>&1
# phase 2: wait for return
for i in $(seq 1 1200); do
  if $B mc info >/dev/null 2>&1; then echo "$(ts) BMC back after ${i}s" >> $M; break; fi
  sleep 2
done
ip neigh show 192.168.1.228 >> $M 2>&1
$B chassis status >> $M 2>&1
# phase 3: attach SOL immediately and record
$B sol deactivate >/dev/null 2>&1
sleep 1
echo "$(ts) attaching SOL" >> $M
setsid $B sol activate > 11-ac-sol.log 2>&1 &
SP=$!
echo "sol_pid=$SP" >> $M
sleep 900
kill $SP 2>/dev/null
$B sol deactivate >/dev/null 2>&1
echo "$(ts) capture ended" >> $M
wc -c 11-ac-sol.log >> $M
$B sel elist 2>&1 | tail -25 > 12-ac-sel.txt
