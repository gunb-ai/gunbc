}

fn required_ci_encode_record(fields: List<String>) -> String {
  concat(concat("r", concat(to_string(length(xs: fields)), ":")), join(fields, ""))
}

fn required_ci_sorted_strings(xs: List<String>) -> List<String> {
  sort_by(xs, fn(s) { s })
}

  required_ci_encode_str(s: layout_tree_directory(tree: tree))
}

fn required_ci_namespace_root_digest(root: LayoutNamespaceRoot) -> String {
  required_ci_encode_record(fields: [
    required_ci_encode_field(name: "namespace", encoded_value: required_ci_encode_str(s: root.namespace)),
    required_ci_encode_field(name: "tree", encoded_value: required_ci_layout_tree_digest(tree: root.tree)),
    required_ci_encode_field(name: "elides", encoded_value: if root.elides_namespace { "t" } else { "f" }),
  ])
}

    required_ci_encode_field(name: "default_tree", encoded_value: required_ci_layout_tree_digest(tree: default_tree)),
    required_ci_encode_field(
      name: "governed_roots",
      encoded_value: required_ci_encode_list(

      )
    ),

      name: "namespace_roots",

  ])
