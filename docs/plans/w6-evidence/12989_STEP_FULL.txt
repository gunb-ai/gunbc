fn compile_pool_slice_install_step(
  node: ComputeHost,
  slice_unit: NonEmptyStr,
  sized: ByteSize,
  slice_unit_file: SystemdUnitFile,
  transport: HostEffectTransport,
  effect_intent: HostEffectIntent,
) -> Reconciliation<HostEffectIntent, HostEffectEvidence> {
  match compile_pool_slice_install_staging(slice_unit: slice_unit) {
    CompilePoolInstallRefused { reason: r } =>
      NotConverged {
        reason: join(
          [
            "host_effect_apply: EnsureCompilePoolSlice refused at staging before any operation ran -- ",
            r as String,
          ],
          "",
        ),
        applied: effect_intent,
      }
    CompilePoolInstallReady { stage_write: write, install: install, daemon_reload: reload, start: start, discard_stage: discard } => {
      let written = run_argv_stdin_over_host_effect_transport(
        command: write,
        stdin_payload: serialize_systemd_unit_file(file: slice_unit_file),
        transport: transport,
      )
      if !run_argv_transport_succeeded(outcome: written) {
        let discarded = run_argv_over_host_effect_transport(command: discard, transport: transport)
        compile_pool_install_failed_step_result(
          failed_step: "stage write",
          slice_unit: slice_unit,
          observed: written,
          discarded: discarded,
          effect_intent: effect_intent,
        )
      } else {
        let installed = run_argv_over_host_effect_transport(command: install, transport: transport)
        if !run_argv_transport_succeeded(outcome: installed) {
          let discarded = run_argv_over_host_effect_transport(command: discard, transport: transport)
          compile_pool_install_failed_step_result(
            failed_step: "install",
            slice_unit: slice_unit,
            observed: installed,
            discarded: discarded,
            effect_intent: effect_intent,
          )
        } else {
          let reloaded = run_argv_over_host_effect_transport(command: reload, transport: transport)
          if !run_argv_transport_succeeded(outcome: reloaded) {
            let discarded = run_argv_over_host_effect_transport(command: discard, transport: transport)
            compile_pool_install_failed_step_result(
              failed_step: "daemon-reload",
              slice_unit: slice_unit,
              observed: reloaded,
              discarded: discarded,
              effect_intent: effect_intent,
            )
          } else {
            let started = run_argv_over_host_effect_transport(command: start, transport: transport)
            let discarded = run_argv_over_host_effect_transport(command: discard, transport: transport)
            match started {
              RunArgvTransportRefused { reason: _ } =>
                compile_pool_install_failed_step_result(
                  failed_step: "start",
                  slice_unit: slice_unit,
                  observed: started,
                  discarded: discarded,
                  effect_intent: effect_intent,
                )
              RunArgvTransportObserved { observed: s } =>
                if !s.success {
                  compile_pool_install_failed_step_result(
                    failed_step: "start",
                    slice_unit: slice_unit,
                    observed: started,
                    discarded: discarded,
                    effect_intent: effect_intent,
                  )
                } else {
                  let after = compile_pool_ensure_admission_for(node: node, transport: transport)
                  if compile_pool_ensure_is_converged(admission: after) {
                    Converged {
                      evidence: ShellEffectApplied {
                        exit_code: s.exit_code,
                        stdout: join(
                          [
                            "CompilePoolSliceReceipt: converged unit=",
                            slice_unit as String,
                            " memory.max=memory.high=",
                            to_string(byte_size_count(b: sized)),
                            " (re-observed after install; the admission answered already-converged)",
                          ],
                          "",
                        ),
                        stderr: s.stderr,
                      },
                      applied: effect_intent,
                    }
                  } else {
                    NotConverged {
                      reason: join(
                        [
                          "host_effect_apply: EnsureCompilePoolSlice installed ",
                          slice_unit as String,
                          " and the read-back did not converge -- ",
                          compile_pool_admission_diagnostic(admission: after),
                        ],
                        "",
                      ),
                      applied: effect_intent,
                    }

fn compile_pool_install_failed_step_result(
  failed_step: String,
  slice_unit: NonEmptyStr,
  observed: RunArgvTransportOutcome,
  discarded: RunArgvTransportOutcome,
  effect_intent: HostEffectIntent,
) -> Reconciliation<HostEffectIntent, HostEffectEvidence> {
  NotConverged {
    reason: join(
      [
        compile_pool_install_step_failure_line(step: failed_step, slice_unit: slice_unit, outcome: observed),
        compile_pool_install_discard_line(outcome: discarded),
      ],
      "",
    ),
    applied: effect_intent,
  }
}


fn compile_pool_admission_diagnostic(admission: EnsureCompilePoolAdmission) -> String {
  match admission {
    EnsureCompilePoolAlreadyConverged { slice_unit: u, sized_bytes: _ } =>
      join(["the slice ", u as String, " is converged"], "")
    EnsureCompilePoolAdmitted { slice_unit: u, sized_bytes: _, slice_unit_file: _ } =>
      join(
        ["the slice ", u as String, " is still admitted for writing, so the installed unit did not take effect on the host"],
        "",
      )
    EnsureCompilePoolRefused { reason: r } => compile_pool_ensure_refusal_detail(reason: r)
  }
}
