// THE STEP-LEVEL FAILURE LINE, one per op so a failed fold names the op that failed rather than
// leaving five exit codes to disambiguate. The old text said "the shell exited"; there is no shell
// on this edge anymore, so the line says the OPERATION exited and which one.
fn compile_pool_install_stage_failure(step: String, slice_unit: NonEmptyStr, observed: ShellExecObserved) -> String {
  join(
    [
      "host_effect_apply: EnsureCompilePoolSlice ran its ",
      step,
      " over ",
      slice_unit as String,
      " and the operation exited ",
      to_string(observed.exit_code),
      ": ",
      if observed.stderr == "" { observed.stdout } else { observed.stderr },
    ],
    "",
  )
}
