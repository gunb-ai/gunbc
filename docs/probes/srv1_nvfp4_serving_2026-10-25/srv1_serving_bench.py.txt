#!/usr/bin/env python3
"""srv1 serving benchmark harness (read-only against the live server).

Measures, for a fixed eval prompt set + load probes, against the running
vLLM server: TTFT, prefill tok/s, decode tok/s, aggregate throughput,
peak GPU memory, and max context that fits.

Credentials: reads the pool key from a file path given by argv; NEVER prints it.
Outputs one JSON-lines receipt to stdout and a JSON summary file.

Usage: srv1_serving_bench.py <key_file> <out_json> [--arm NAME] [--skip-longctx]
"""
import http.client
import json
import os
import ssl
import statistics
import sys
import threading
import time

HOST = "127.0.0.1"
PORT = 18000
MODEL = "Qwen3.8-Flash-Next"

# ---------------------------------------------------------------- prompts
# Fixed eval prompt set: deterministic (temperature=0), short, coding/reasoning
# flavored, sized for correctness diffing across KV-dtype arms.
EVAL_PROMPTS = [
    {"id": "code-rot13",
     "prompt": "Write a Python function `rot13(s)` that returns the ROT13 of an ASCII string, leaving non-letters unchanged. Return only the code.",
     "max_tokens": 160},
    {"id": "code-bugfix",
     "prompt": "This Python has a bug. Identify it in one sentence, then give the fixed line.\n```\ndef mean(xs):\n    return sum(xs) / len(xs) - 1\n```",
     "max_tokens": 120},
    {"id": "reason-train",
     "prompt": "A train leaves city A at 9:00 going 80 km/h. Another leaves city B at 10:00 going 120 km/h toward A. The cities are 400 km apart. When do they meet? Show the arithmetic briefly, then state the time.",
     "max_tokens": 200},
    {"id": "recall-facts",
     "prompt": "Remember these pairs: ALPHA=42, BRAVO=7, CHARLIE=99. Answer only: what is CHARLIE minus BRAVO?",
     "max_tokens": 32},
    {"id": "json-struct",
     "prompt": 'Output a JSON object with keys "planets" (a list of the first 4 planets from the Sun, in order) and "count" (their number). JSON only, no prose.',
     "max_tokens": 80},
    {"id": "code-tests",
     "prompt": "Write pytest test cases for a function `is_palindrome(s)` covering empty string, single char, 'racecar', and 'hello'. Return only code.",
     "max_tokens": 220},
]

FILLER = "The quick brown fox jumps over the lazy dog while the engineer reviews the pull request. "


def ctx_probe_prompt(target_tokens):
    # Calibrated on this tokenizer+server (2026-10-25): FILLER repeated to
    # 180_000 CHARS measured 34_403 prompt tokens => 0.1911 tok/char
    # (5.23 chars/token; highly repetitive text).
    chars = int(target_tokens / 0.1911)
    reps = chars // len(FILLER) + 2
    body = (FILLER * reps)[:chars]
    return body + "\n\nQuestion: How many times does the word 'fox' appear? Answer with just the number."


LONGCTX_PROBES = [
    {"id": "ctx-120k", "target_tokens": 120_000},
    {"id": "ctx-180k", "target_tokens": 180_000},   # the premium target
    {"id": "ctx-240k", "target_tokens": 240_000},
    {"id": "ctx-260k", "target_tokens": 260_000},   # max_model_len edge
]

# ---------------------------------------------------------------- transport
_key = None


def get_key(key_file):
    global _key
    if _key is None:
        with open(key_file) as f:
            _key = f.read().strip()
    return _key


def _conn():
    ctx = ssl._create_unverified_context()
    return http.client.HTTPSConnection(HOST, PORT, context=ctx, timeout=1800)


def post(path, payload, stream=False):
    """POST JSON. If stream, yields decoded chunks; else returns (status, obj)."""
    body = json.dumps(payload).encode()
    c = _conn()
    c.request("POST", path, body=body, headers={
        "Authorization": "Bearer " + get_key(sys.argv[1]),
        "Content-Type": "application/json",
    })
    r = c.getresponse()
    if not stream:
        data = r.read().decode("utf-8", "replace")
        obj = None
        try:
            obj = json.loads(data)
        except Exception:
            pass
        out = (r.status, obj, data[:600] if obj is None else "")
        c.close()
        return out
    return c, r


def stream_chat(payload):
    """Return (ttft_s, total_s, usage, text, err). Captures content + reasoning
    deltas (this checkpoint streams thinking into `delta.reasoning`)."""
    payload = dict(payload, stream=True,
                   stream_options={"include_usage": True})
    t0 = time.perf_counter()
    ttft = None
    text = []
    reasoning = []
    usage = None
    err = None
    try:
        c, r = post("/v1/chat/completions", payload, stream=True)
    except Exception as e:  # noqa
        return None, None, None, "", f"connect:{e}"
    if r.status != 200:
        body = r.read().decode("utf-8", "replace")[:400]
        c.close()
        return None, None, None, "", f"http{r.status}:{body}"
    try:
        buf = ""
        while True:
            chunk = r.read(4096)
            if not chunk:
                break
            buf += chunk.decode("utf-8", "replace")
            while "\n" in buf:
                line, buf = buf.split("\n", 1)
                if not line.startswith("data:"):
                    continue
                data = line[5:].strip()
                if data == "[DONE]":
                    continue
                try:
                    o = json.loads(data)
                except Exception:
                    continue
                if o.get("usage"):
                    usage = o["usage"]
                for ch in o.get("choices", []):
                    d = ch.get("delta") or {}
                    tok = d.get("content")
                    if tok is None:
                        tok = d.get("reasoning")
                        if tok:
                            reasoning.append(tok)
                    if tok:
                        if ttft is None:
                            ttft = time.perf_counter() - t0
                        if d.get("content") is not None:
                            text.append(tok)
    except Exception as e:  # noqa
        err = f"stream:{e}"
    total = time.perf_counter() - t0
    c.close()
    full = "".join(reasoning) + "\n---ANSWER---\n" + "".join(text)
    return ttft, total, usage, full, err


def completions_raw(prompt, max_tokens):
    """Non-stream /v1/completions for context probes. Return dict."""
    payload = {"model": MODEL, "prompt": prompt,
               "max_tokens": max_tokens, "temperature": 0}
    t0 = time.perf_counter()
    status, obj, raw = post("/v1/completions", payload)
    total = time.perf_counter() - t0
    if status != 200 or obj is None:
        return {"status": status, "error": raw or (obj or {}).get("error"),
                "total_s": total}
    try:
        u = obj.get("usage", {})
        return {"status": status, "total_s": total,
                "prompt_tokens": u.get("prompt_tokens"),
                "completion_tokens": u.get("completion_tokens"),
                "text": obj["choices"][0]["text"][:120]}
    except Exception as e:  # noqa
        return {"status": status, "error": f"parse:{e}", "total_s": total}


# ---------------------------------------------------------------- gpu sampler
class GpuSampler(threading.Thread):
    def __init__(self):
        super().__init__(daemon=True)
        self.stop_flag = threading.Event()
        self.samples = []

    def run(self):
        import subprocess
        while not self.stop_flag.is_set():
            try:
                out = subprocess.run(
                    ["nvidia-smi", "--query-gpu=memory.used,utilization.gpu",
                     "--format=csv,noheader,nounits"],
                    capture_output=True, text=True, timeout=5).stdout.strip()
                mem, util = [int(x) for x in out.split(",")]
                self.samples.append((time.time(), mem, util))
            except Exception:
                pass
            self.stop_flag.wait(2.0)

    def peak(self):
        if not self.samples:
            return None, None
        return (max(s[1] for s in self.samples),
                max(s[2] for s in self.samples))


# ---------------------------------------------------------------- phases
def run_eval(arm):
    rows = []
    for p in EVAL_PROMPTS:
        msgs = [{"role": "user", "content": p["prompt"]}]
        ttft, total, usage, text, err = stream_chat(
            {"model": MODEL, "messages": msgs,
             "max_tokens": p["max_tokens"], "temperature": 0})
        row = {"phase": "eval", "arm": arm, "id": p["id"], "error": err}
        if usage and ttft is not None:
            pt = usage.get("prompt_tokens", 0)
            ct = usage.get("completion_tokens", 0)
            cached = (usage.get("prompt_tokens_details") or {}).get("cached_tokens")
            row.update({
                "prompt_tokens": pt, "completion_tokens": ct,
                "cached_tokens": cached,
                "ttft_s": round(ttft, 3), "total_s": round(total, 3),
                "prefill_tps": round(pt / ttft, 1) if ttft else None,
                "decode_tps": round(ct / (total - ttft), 1) if (total - ttft) > 0.01 else None,
                "output_sha": __import__("hashlib").sha256(text.encode()).hexdigest()[:16],
                "answer_sha": __import__("hashlib").sha256(text.split("---ANSWER---")[-1].encode()).hexdigest()[:16],
                "output_head": text[:160],
                "answer_head": (text.split("---ANSWER---")[-1].strip())[:200],
            })
        rows.append(row)
        print(json.dumps(row), flush=True)
    return rows


def run_decode_heavy(arm):
    """Single-stream long decode: measures sustained decode tok/s with MTP on."""
    msgs = [{"role": "user", "content":
             "Count down from 200 to 1, one number per line. Output only the numbers."}]
    ttft, total, usage, text, err = stream_chat(
        {"model": MODEL, "messages": msgs, "max_tokens": 700, "temperature": 0})
    row = {"phase": "decode-heavy", "arm": arm, "id": "countdown-200", "error": err}
    if usage and ttft is not None:
        ct = usage.get("completion_tokens", 0)
        row.update({"completion_tokens": ct, "ttft_s": round(ttft, 3),
                    "total_s": round(total, 3),
                    "decode_tps": round(ct / (total - ttft), 1)})
    print(json.dumps(row), flush=True)
    return [row]


def run_prefill_heavy(arm):
    """Prefill-dominated: long prompt, tiny completion -> prefill tok/s."""
    prompt = ctx_probe_prompt(30_000)
    msgs = [{"role": "user", "content": prompt}]
    ttft, total, usage, text, err = stream_chat(
        {"model": MODEL, "messages": msgs, "max_tokens": 8, "temperature": 0})
    row = {"phase": "prefill-heavy", "arm": arm, "id": "prefill-30k", "error": err}
    if usage and ttft is not None:
        pt = usage.get("prompt_tokens", 0)
        row.update({"prompt_tokens": pt, "ttft_s": round(ttft, 3),
                    "prefill_tps": round(pt / ttft, 1)})
    print(json.dumps(row), flush=True)
    return [row]


def run_burst(arm, n=16):
    """Concurrent aggregate throughput."""
    import concurrent.futures as cf
    def one(i):
        msgs = [{"role": "user", "content": f"Reply with exactly: line {i} of 16. Nothing else."}]
        return stream_chat({"model": MODEL, "messages": msgs,
                            "max_tokens": 24, "temperature": 0})
    t0 = time.perf_counter()
    with cf.ThreadPoolExecutor(max_workers=n) as ex:
        res = list(ex.map(one, range(1)))  if False else list(ex.map(one, range(n)))
    wall = time.perf_counter() - t0
    ct = sum((u or {}).get("completion_tokens", 0) for _, _, u, _, e in res if not e)
    ttfts = [t for t, _, _, _, e in res if t and not e]
    row = {"phase": "burst", "arm": arm, "id": f"burst-{n}",
           "errors": sum(1 for *_, e in res if e),
           "wall_s": round(wall, 2), "total_completion_tokens": ct,
           "aggregate_decode_tps": round(ct / wall, 1) if wall else None,
           "ttft_p50_s": round(statistics.median(ttfts), 3) if ttfts else None,
           "ttft_max_s": round(max(ttfts), 3) if ttfts else None}
    print(json.dumps(row), flush=True)
    return [row]


def run_longctx(arm):
    rows = []
    for p in LONGCTX_PROBES:
        r = completions_raw(ctx_probe_prompt(p["target_tokens"]), 16)
        r.update({"phase": "longctx", "arm": arm, "id": p["id"],
                  "target_tokens": p["target_tokens"]})
        rows.append(r)
        print(json.dumps(r), flush=True)
    return rows


def main():
    key_file, out_path = sys.argv[1], sys.argv[2]
    arm = "fp8_kv_flash_attn"
    if "--arm" in sys.argv:
        arm = sys.argv[sys.argv.index("--arm") + 1]
    skip_long = "--skip-longctx" in sys.argv
    assert key_file and not key_file.startswith("-")

    gpu = GpuSampler()
    gpu.start()
    summary = {"arm": arm, "date_utc": time.strftime("%Y-%m-%dT%H:%M:%SZ", time.gmtime()),
               "host": os.uname().nodename, "engine": f"https://{HOST}:{PORT}",
               "rows": []}
    summary["rows"] += run_eval(arm)
    summary["rows"] += run_prefill_heavy(arm)
    summary["rows"] += run_decode_heavy(arm)
    summary["rows"] += run_burst(arm)
    if not skip_long:
        summary["rows"] += run_longctx(arm)
    gpu.stop_flag.set()
    peak_mem, peak_util = gpu.peak()
    summary["peak_gpu_mem_mib"] = peak_mem
    summary["peak_gpu_util_pct"] = peak_util
    with open(out_path, "w") as f:
        json.dump(summary, f, indent=1)
    print("WROTE " + out_path, flush=True)


if __name__ == "__main__":
    main()
